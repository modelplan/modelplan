type u14RecordingTransport struct{}

func u14HTTPClient() *http.Client {

}

// Match this scenario's fixed synthetic inputs independently of the production
// guard. Exact fixture bodies are stricter than the approved Gemini schema;
// production transport tests separately cover every legal selector/body shape.
func u14POSTMatches(r *http.Request, expectedBody string, credentials http.Header) (bool, bool, bool) {
	body, err := io.ReadAll(io.LimitReader(r.Body, 4097))
	r.Body.Close()
	bodyOK := err == nil && bytes.Equal(body, []byte(expectedBody)) && r.ContentLength == int64(len(body)) && len(r.TransferEncoding) == 0
	expected := credentials.Clone()
	expected.Set("Accept", "application/json")
	expected.Set("Content-Type", "application/json")
	expected.Set("User-Agent", "modelplan")
	headersOK := len(r.Header) == len(expected)
	for name, values := range r.Header {
		if len(values) != 1 || r.Header.Get(name) != expected.Get(name) {
			headersOK = false
		}
	}
	credentialOK := true
	for name := range credentials {
		if r.Header.Get(name) != credentials.Get(name) {
			credentialOK = false
		}
	}
	return bodyOK, headersOK, credentialOK
}
